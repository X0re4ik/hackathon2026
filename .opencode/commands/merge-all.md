# Оркестратор батч-процессинга

Твоя единственная задача — выполнить приведенный ниже bash-скрипт в терминале. Ничего не меняй в скрипте. 
Этот скрипт обойдет все файлы `META-*.md` и для каждого поднимет изолированный процесс субагента `jobs-merger`, предотвращая утечку контекста и состояния гонки.

Выполни этот код:

```bash
#!/bin/bash
META_DIR="output/meta"

# Проверяем, есть ли файлы
if ! ls "$META_DIR"/META-*.md 1> /dev/null 2>&1; then
    echo "Файлы метаданных не найдены в $META_DIR"
    exit 1
fi

# 1. Цикл по файлам
for file in "$META_DIR"/META-*.md; do
    echo -e "\n============================================="
    echo "▶ ЗАПУСК ИТЕРАЦИИ: $file"
    echo "============================================="
    
    # Вызов субагента в неинтерактивном режиме
    opencode run --agent jobs-merger "Выполни строго одну итерацию слияния. Твой входной файл: $file. Строго следуй правилам скилла и не трогай другие файлы."
    
    # Опциональная пауза, чтобы снизить риск rate limit по API
    sleep 2
done

# 2. Финализация
echo -e "\n============================================="
echo "▶ ВСЕ ИТЕРАЦИИ ЗАВЕРШЕНЫ. ЗАПУСК ФИНАЛИЗАЦИИ."
echo "============================================="
opencode run --agent jobs-merger "Все меты обработаны. Прочитай finalization.md в папке скилла и выполни финализацию файла свода."
```
